1""" 

2Cache em memória, por tempo, das respostas de BI (guia, seção 5.5). 

3 

4As respostas dependem só dos filtros e do estado do SGI no momento; repetir a 

5mesma agregação a cada clique no portal só carrega o banco transacional da 

6operação. Um cache de `BI_CACHE_TTL_SECONDS` (60 s por padrão) por combinação 

7de parâmetros resolve isso. `0` desliga. 

8 

9É um dicionário protegido por lock, por processo: com mais de um worker do 

10uvicorn cada um tem o seu, o que é aceitável para uma janela de 60 s. 

11""" 

12 

13import threading 

14import time 

15from collections.abc import Callable, Hashable 

16from typing import Any 

17 

18from app.core.config import settings 

19 

20#: Acima disto a gravação descarta as entradas vencidas antes de inserir. 

21_LIMPEZA_A_PARTIR_DE = 256 

22 

23 

24class TtlCache: 

25 def __init__(self, ttl_seconds: float, clock: Callable[[], float] = time.monotonic) -> None: 

26 self.ttl = ttl_seconds 

27 self._clock = clock 

28 self._lock = threading.Lock() 

29 self._items: dict[Hashable, tuple[float, Any]] = {} 

30 

31 @property 

32 def enabled(self) -> bool: 

33 return self.ttl > 0 

34 

35 def get(self, key: Hashable) -> Any | None: 

36 if not self.enabled: 

37 return None 

38 with self._lock: 

39 item = self._items.get(key) 

40 if item is None: 

41 return None 

42 expires_at, value = item 

43 if expires_at <= self._clock(): 

44 del self._items[key] 

45 return None 

46 return value 

47 

48 def set(self, key: Hashable, value: Any) -> None: 

49 if not self.enabled: 

50 return 

51 with self._lock: 

52 if len(self._items) >= _LIMPEZA_A_PARTIR_DE: 

53 agora = self._clock() 

54 for vencida in [k for k, (fim, _) in self._items.items() if fim <= agora]: 

55 del self._items[vencida] 

56 self._items[key] = (self._clock() + self.ttl, value) 

57 

58 def get_or_build(self, key: Hashable, build: Callable[[], Any]) -> Any: 

59 """Devolve o valor em cache ou constrói, guarda e devolve o novo.""" 

60 cached = self.get(key) 

61 if cached is not None: 

62 return cached 

63 value = build() 

64 self.set(key, value) 

65 return value 

66 

67 def clear(self) -> None: 

68 with self._lock: 

69 self._items.clear() 

70 

71 def __len__(self) -> int: 

72 with self._lock: 

73 return len(self._items) 

74 

75 

76_cache: TtlCache | None = None 

77 

78 

79def bi_cache() -> TtlCache: 

80 """Cache compartilhado dos serviços de BI, criado a partir das settings.""" 

81 global _cache 

82 if _cache is None: 

83 _cache = TtlCache(settings.BI_CACHE_TTL_SECONDS) 

84 return _cache 

85 

86 

87def reset_bi_cache() -> None: 

88 """Descarta o cache compartilhado (testes e troca de configuração).""" 

89 global _cache 

90 _cache = None
