1""" 

2Auxiliares genéricos compartilhados pelos serviços. 

3 

4Aqui mora o que mais de um módulo de `app/services` usa e não pertence a 

5nenhum deles em particular: normalização de texto, o relógio da operação, 

6arredondamento e o ambiente Jinja2 dos templates. Nada de catálogo, de regra 

7de negócio nem de banco — o que depende de um domínio continua no módulo do 

8domínio. 

9 

10Antes desta camada o mesmo código existia duas vezes (`texto` e `_texto`) e 

11módulos sem relação entre si se importavam só para pegar um auxiliar: 

12`commercial_overview` pegava o relógio em `os_overview`, `os_aggregates` e 

13`commercial_catalog` pegavam a normalização de texto no catálogo de OS, e o 

14serviço de e-mail e o de relatórios montavam cada um o seu Jinja2 apontando 

15para a mesma pasta. Quem precisa de um auxiliar importa daqui. 

16"""

18import math 

19import re 

20import unicodedata 

21from datetime import date, datetime 

22from pathlib import Path 

23from typing import Any 

24from zoneinfo import ZoneInfo

28# --- Texto --------------------------------------------------------------------- 

29 

30_ESPACOS = re.compile(r"\s+") 

31 

32 

33def texto(valor: Any) -> str | None: 

34 """Valor do banco como `str`, preservando o `None` (UUID, enum, Decimal...).""" 

35 return None if valor is None else str(valor) 

36 

37 

38def collapse_spaces(value: str) -> str: 

39 """Espaço repetido vira um só; sobra nas pontas some.""" 

40 return _ESPACOS.sub(" ", value).strip() 

41 

42 

43def strip_accents(value: str) -> str: 

44 """`Instalação` -> `Instalacao`: decompõe e descarta os diacríticos.""" 

45 decomposto = unicodedata.normalize("NFKD", value) 

46 return "".join(ch for ch in decomposto if not unicodedata.combining(ch)) 

47 

48 

49def norm_key(value: str) -> str: 

50 """Chave de comparação tolerante: sem acento, sem caixa, sem espaços extras.""" 

51 return collapse_spaces(strip_accents(value)).casefold() 

52 

53 

54def ptbr_sort_key(value: str) -> tuple[str, str]: 

55 """Aproximação do localeCompare pt-BR do portal: sem acento e sem caixa primeiro.""" 

56 return (strip_accents(value).casefold(), value)

59# --- Data e hora --------------------------------------------------------------- 

60 

61#: Fuso da operação: as datas do contrato são "dia local", nunca UTC. 

62TIMEZONE = ZoneInfo("America/Sao_Paulo") 

63 

64 

65def now_local() -> datetime: 

66 return datetime.now(TIMEZONE) 

67 

68 

69def today_local() -> date: 

70 return now_local().date() 

71 

72 

73def local_date(momento: datetime | None) -> date | None: 

74 """ 

75 Dia em America/Sao_Paulo de um timestamp do SGI. 

76 

77 Timestamp com fuso é convertido; sem fuso já é hora local do SGI 

78 (`ANALISE-BANCO.md` §8.5) e só perde a hora. 

79 """ 

80 if momento is None: 

81 return None 

82 if momento.tzinfo is None: 

83 return momento.date() 

84 return momento.astimezone(TIMEZONE).date() 

85 

86 

87def local_datetime(momento: datetime | None) -> datetime | None: 

88 """ 

89 Momento do SGI na hora da operação, com o fuso explícito na resposta. 

90 

91 Mesma regra do `local_date`, guardando a HORA: é o que a página de 

92 Técnicos precisa para montar a jornada do dia e a curva de horário. Sem 

93 fuso, o timestamp já é hora local do SGI e só ganha o rótulo do fuso; com 

94 fuso, é convertido. O contrato sai em ISO com o deslocamento escrito 

95 (`-03:00`), para o navegador nunca ter que adivinhar. 

96 """ 

97 if momento is None: 

98 return None 

99 if momento.tzinfo is None: 

100 return momento.replace(tzinfo=TIMEZONE) 

101 return momento.astimezone(TIMEZONE)

107def round_half_up(value: float, ndigits: int = 0) -> float: 

108 """`Math.round` do JavaScript para valores não negativos (meio para cima).""" 

109 factor = 10**ndigits 

110 return math.floor(value * factor + 0.5) / factor 

111 

112 

113def money(value: float) -> float: 

114 """Soma de `double precision` sai com ruído binário; centavos são o grão real.""" 

115 return round(value, 2)

118# --- Templates ----------------------------------------------------------------- 

119 

120#: Raiz dos templates Jinja2. O e-mail e o PDF do relatório saem dos mesmos 

121#: arquivos em `app/templates/`; o PDF ainda precisa do caminho para resolver 

122#: as imagens e o CSS relativos (`base_url` do WeasyPrint). 

123TEMPLATES_DIR = Path(__file__).resolve().parents[1] / "templates" 

124 

125_jinja = Environment( 

126 loader=FileSystemLoader(TEMPLATES_DIR), 

127 autoescape=select_autoescape(["html", "xml"]), 

128) 

129 

130 

131def render_template(template: str, context: dict[str, Any]) -> str: 

132 """Renderiza um template relativo a `app/templates/`, com autoescape de HTML.""" 

133 return _jinja.get_template(template).render(**context)
