73 code: str | None

102@dataclass(frozen=True) 

103class RawCoverage: 

104 """Cobertura da base inteira: limites de agendamento e última alteração.""" 

105 

106 min_date: date | None 

107 max_date: date | None 

108 updated_at: datetime | None 

109 

110 

111# `service_orders.address` é json com o endereço da OS; `customer_addresses` 

112# (via assinatura) é o fallback de cidade para OS sem endereço próprio. O nome 

113# do técnico entra só para derivar a equipe. Soft delete sempre respeitado. 

114# 

115# FUSO, achado de 21/09/2026 que decide o formato destas colunas: até 

116# 01/09/2026 o SGI gravou `started_motion_at` e `arrived_at` TRÊS HORAS 

117# adiantados, e corrigiu em 02/09. Medido em produção: a mediana de 

118# `arrived_at - started_at` era 179,6 min (3h exatas) até 01/09 e passou a 0 

119# depois. `started_at` e `finished_at` nunca erraram. 

120# Consequência para o contrato: a DURAÇÃO do deslocamento é correta na base 

121# inteira, porque os dois marcos erraram juntos, mas o INSTANTE de cada um não 

122# é. Por isso o deslocamento sai daqui já em minutos e os dois horários crus 

123# não saem: o erro morre no serviço e nunca chega a uma tela. 

124_RECORDS_SQL = """ 

125SELECT 

126 so.id::text AS id, 

127 so.code AS code, 

128 so.scheduled_date AS scheduled_date, 

129 so.period AS period_key, 

130 so.technician_id::text AS technician_id, 

131 tech.name AS technician_name, 

132 -- Equipes do técnico no cadastro do SGI. Subconsulta escalar de propósito: 

133 -- um JOIN em team_users duplicaria a OS quando o técnico está em duas 

134 -- equipes (a tabela não tem unicidade por usuário). 

135 (SELECT array_agg(t.name ORDER BY t.name) 

136 FROM public.team_users tu 

137 JOIN public.teams t ON t.id = tu.team_id AND t.deleted_at IS NULL 

138 WHERE tu.user_id = so.technician_id) AS technician_team_names, 

139 so.type AS type_key, 

140 soc.name AS category, 

141 so.status AS status_key, 

142 sor.label AS resolution_raw, 

143 so.address AS address, 

144 ca.city AS subscription_city, 

145 so.started_at AS started_at, 

146 so.finished_at AS finished_at, 

147 EXTRACT(EPOCH FROM (so.arrived_at - so.started_motion_at)) / 60.0 AS travel_minutes, 

148 so.difficulty_level AS difficulty_key, 

149 so.unproductive_reason AS unproductive_reason_raw 

150FROM public.service_orders so 

151LEFT JOIN auth.users tech ON tech.id = so.technician_id 

152LEFT JOIN public.service_order_categories soc ON soc.id = so.category_id 

153LEFT JOIN public.service_order_resolution sor ON sor.id = so.resolution_id 

154LEFT JOIN public.subscriptions s ON s.id = so.subscription_id 

155LEFT JOIN public.customer_addresses ca ON ca.id = s.customer_address_id 

156WHERE so.deleted_at IS NULL 

157 AND so.scheduled_date IS NOT NULL 

158 {filtros} 

159ORDER BY so.scheduled_date, so.created_at, so.code 

160""" 

161 

162# Cadastro de cidades do SGI: a linha-pai é o rótulo do tipo ("cities" -> 

163# "Cidades"), então só as filhas interessam. 

164_CITY_REGISTRY_SQL = text( 

165 """ 

166SELECT et.key AS grafia, 

167 et.value AS cidade 

168FROM public.enum_types et 

169WHERE et.type = :tipo 

170 AND et.parent_id IS NOT NULL 

171 AND et.deleted_at IS NULL 

172""" 

173) 

174 

175_COVERAGE_SQL = text( 

176 """ 

177SELECT 

178 MIN(so.scheduled_date) AS min_date, 

179 MAX(so.scheduled_date) AS max_date, 

180 MAX(GREATEST(so.updated_at, so.created_at)) AS updated_at 

181FROM public.service_orders so 

182WHERE so.deleted_at IS NULL 

183 AND so.scheduled_date IS NOT NULL 

184"""

190) -> list[RawServiceOrder]: 

191 """OS agendadas dentro da janela (limites inclusivos; `None` = sem limite).""" 

192 condicoes: list[str] = [] 

193 params: dict[str, Any] = {} 

194 if date_from is not None: 

195 condicoes.append("AND so.scheduled_date >= :date_from") 

196 params["date_from"] = date_from 

197 if date_to is not None: 

198 condicoes.append("AND so.scheduled_date <= :date_to") 

199 params["date_to"] = date_to 

200 

201 sql = _RECORDS_SQL.format(filtros="\n ".join(condicoes)) 

202 linhas = conn.execute(text(sql), params).mappings().all() 

203 return [ 

204 RawServiceOrder( 

205 id=str(linha["id"]), 

206 code=texto(linha["code"]), 

207 scheduled_date=linha["scheduled_date"], 

208 period_key=texto(linha["period_key"]), 

209 technician_id=texto(linha["technician_id"]), 

210 technician_name=texto(linha["technician_name"]), 

211 type_key=texto(linha["type_key"]), 

212 category=texto(linha["category"]), 

213 status_key=texto(linha["status_key"]), 

214 resolution_raw=texto(linha["resolution_raw"]), 

215 address=linha["address"], 

216 subscription_city=texto(linha["subscription_city"]), 

217 technician_team_names=( 

218 tuple(linha["technician_team_names"]) 

219 if linha["technician_team_names"] 

220 else None 

221 ), 

222 started_at=linha["started_at"], 

223 finished_at=linha["finished_at"], 

224 travel_minutes=(

228 ), 

229 difficulty_key=texto(linha["difficulty_key"]), 

230 unproductive_reason_raw=texto(linha["unproductive_reason_raw"]), 

231 ) 

232 for linha in linhas

257def build_records( 

258 rows: list[RawServiceOrder], city_registry: Mapping[str, str] | None = None 

259) -> tuple[list[OsOverviewRecord], OsQuality]: 

260 """ 

261 Aplica o de-para do BI a cada linha crua e junta as pendências encontradas. 

262 

263 Puro (sem banco): é o que os testes cobrem contra os casos aprovados. O 

264 cadastro de cidades entra por parâmetro (`query_city_registry`) justamente 

265 para esta função continuar pura; sem ele, nenhuma grafia é reconhecida e 

266 todas viram pendência, que é o comportamento honesto para um cadastro 

267 ausente. 

268 """ 

269 cadastro_cidades = city_registry if city_registry is not None else {} 

270 registros: list[OsOverviewRecord] = [] 

271 # dicts como conjuntos ordenados: a ordem de aparição ajuda a investigar 

272 tecnicos_sem_equipe: dict[str, None] = {} 

273 status_desconhecidos: dict[str, None] = {} 

274 tipos_desconhecidos: dict[str, None] = {} 

275 cidades_desconhecidas: dict[str, None] = {} 

276 sem_equipe = 0 

277 

278 for linha in rows: 

279 equipe: str | None = None 

280 if linha.technician_id: 

281 # ÚNICA fonte da equipe: o cadastro do SGI (teams/team_users). 

282 # Técnico sem vínculo fica sem equipe e vira pendência — a solução 

283 # é vincular no SGI, nunca uma lista no código (decisão de 

284 # 04/09/2026: nenhum dado da página nasce na aplicação). 

285 equipe = catalog.team_from_sgi_teams(linha.technician_team_names) 

286 if equipe is None: 

287 tecnicos_sem_equipe[linha.technician_id] = None 

288 sem_equipe += 1 

289 

290 tipo, tipo_conhecido = catalog.normalize_service_type(linha.type_key) 

291 if not tipo_conhecido: 

292 tipos_desconhecidos[linha.type_key or ""] = None 

293 

294 status, status_conhecido = catalog.normalize_status(linha.status_key) 

295 if not status_conhecido: 

296 status_desconhecidos[linha.status_key or ""] = None 

297 

298 cidade_crua = catalog.city_from_address(linha.address) or linha.subscription_city 

299 cidade, cidade_conhecida = catalog.city_from_registry(cidade_crua, cadastro_cidades) 

300 if not cidade_conhecida and cidade_crua: 

301 # Grafia fora do cadastro do SGI: a cidade APARECE como foi 

302 # digitada e a pendência fica registrada. Até 08/09/2026 ela virava 

303 # "Cidade não informada" e desaparecia do gráfico. 

304 cidades_desconhecidas[collapse_spaces(cidade_crua)] = None 

305 

306 registros.append( 

307 OsOverviewRecord( 

308 id=linha.code or linha.id, 

309 date=linha.scheduled_date, 

310 period=catalog.normalize_period(linha.period_key), 

311 technician_id=linha.technician_id, 

312 team=equipe, 

313 service_type=tipo, 

314 category=catalog.normalize_category(linha.category), 

315 city=cidade, 

316 status=status, 

317 status_key=linha.status_key, 

318 resolution=catalog.normalize_resolution(linha.resolution_raw), 

319 started_at=local_datetime(linha.started_at), 

320 finished_at=local_datetime(linha.finished_at), 

321 # Deslocamento negativo é marcação errada no app do técnico 

322 # (chegou antes de sair): vira nulo, que é "não medido". Zero 

323 # aqui seria mentira, e entraria na média puxando para baixo. 

324 travel_minutes=( 

325 round(linha.travel_minutes, 1) 

326 if linha.travel_minutes is not None and linha.travel_minutes >= 0 

327 else None 

328 ), 

329 difficulty=catalog.normalize_difficulty(linha.difficulty_key), 

330 unproductive_reason=catalog.normalize_unproductive_reason( 

331 linha.unproductive_reason_raw 

332 ), 

333 ) 

334 ) 

335 

336 qualidade = OsQuality( 

337 records_without_team=sem_equipe, 

338 unclassified_technician_ids=list(tecnicos_sem_equipe), 

339 unmapped_status_keys=list(status_desconhecidos), 

340 unmapped_service_type_keys=list(tipos_desconhecidos), 

341 unknown_cities=list(cidades_desconhecidas), 

342 ) 

343 return registros, qualidade

346def build_catalog( 

347 rows: list[RawServiceOrder], records: list[OsOverviewRecord] 

348) -> OsCatalog: 

349 """ 

350 Técnicos presentes nas linhas, com nome, para o oitavo filtro da página, e 

351 a ordem de exibição de status e tipos (`os_catalog.catalog_entries`). 

352 

353 O nome sai de `auth.users.name` e entra na resposta DE PROPÓSITO: o filtro 

354 de técnico do portal guarda o identificador, e o seletor fica desabilitado 

355 enquanto o catálogo vier vazio (`src/data/api/osOverviewSource.ts`) — o 

356 portal prefere filtro inerte a listar uuid na tela. 

357 

358 Só entram técnicos com OS no lote: isto é o catálogo do recorte, não a 

359 lista de funcionários. Técnico sem nome em `auth.users` fica de fora (o 

360 seletor não teria o que mostrar), mas continua contando nos totais. 

361 """ 

362 nomes: dict[str, str] = {} 

363 for linha in rows: 

364 if not linha.technician_id or not linha.technician_name: 

365 continue 

366 nome = collapse_spaces(linha.technician_name) 

367 if nome: 

368 nomes.setdefault(linha.technician_id, nome) 

369 return OsCatalog( 

370 technicians=[OsTechnician(id=tid, name=nome) for tid, nome in nomes.items()], 

371 statuses=[ 

372 OsCatalogEntry(label=label, order=order) 

373 for label, order in catalog.catalog_entries( 

374 catalog.STATUS_ORDER, (r.status for r in records) 

375 ) 

376 ], 

377 service_types=[ 

378 OsCatalogEntry(label=label, order=order) 

379 for label, order in catalog.catalog_entries( 

380 catalog.SERVICE_TYPE_ORDER, (r.service_type for r in records) 

381 ) 

382 ], 

383 )

410class OsOverviewService: 

411 """ 

412 Consulta o SGI e monta as respostas da Visão Geral de OS. 

413 

414 O acesso ao banco fica isolado em `_load_rows` e `_load_coverage` para os 

415 testes trocarem só eles (dublê via `dependency_overrides`), como o 

416 `FakeClient` do BigQuery. As respostas ficam no cache de BI pelo tempo de 

417 `BI_CACHE_TTL_SECONDS`, por combinação de parâmetros. 

418 """

421 self._cache = cache if cache is not None else bi_cache()

427 return query_coverage(conn)

441 ) -> OsOverviewRecordsResponse:

443 chave = ("os.records", date_from, date_to) 

444 return self._cache.get_or_build(chave, lambda: self._records(date_from, date_to))

446 def _records(self, date_from: date | None, date_to: date | None) -> OsOverviewRecordsResponse:

448 cadastro_cidades = self._load_city_registry()

455 source=OsSource( 

456 updated_at=local_date(cobertura.updated_at), 

457 coverage=OsCoverage(min=cobertura.min_date, max=cobertura.max_date), 

458 ), 

459 filters=OsRecordsFilters(date_from=date_from, date_to=date_to), 

460 count=len(registros), 

461 records=registros, 

462 catalog=build_catalog(linhas, registros), 

463 quality=qualidade, 

464 )

498 def _overview( 

499 self, 

500 date_from: date | None, 

501 date_to: date | None, 

502 team: str | None, 

503 technician_id: str | None, 

504 service_type: str | None, 

505 category: str | None, 

506 city: str | None, 

507 status: str | None, 

508 ) -> OsOverviewResponse: 

509 cobertura = self._load_coverage() 

510 coverage = OsCoverage(min=cobertura.min_date, max=cobertura.max_date) 

511 

512 # Sem período: o padrão do portal (mês da data máxima da base até ela). 

513 if date_from is None or date_to is None: 

514 padrao_inicio, padrao_fim = default_period(coverage, today_local()) 

515 date_from = date_from if date_from is not None else padrao_inicio 

516 date_to = date_to if date_to is not None else padrao_fim 

517 _validar_ordem(date_from, date_to) 

518 

519 # A janela carregada inclui o período anterior de mesma duração, para 

520 # os deltas saírem do mesmo lote de registros normalizados. 

521 janela_inicio, _ = previous_same_length(date_from, date_to) 

522 cadastro_cidades = self._load_city_registry() 

523 linhas = self._load_rows(janela_inicio, date_to) 

524 registros, qualidade = build_records(linhas, cadastro_cidades) 

525 log_quality("agregados", qualidade) 

526 

527 filtros = OsFilters( 

528 date_from=date_from, 

529 date_to=date_to,
