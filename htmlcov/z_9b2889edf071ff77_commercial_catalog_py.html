99def city_for(raw: str | None) -> CityMatch: 

100 """ 

101 Id e rótulo comercial de uma grafia de `customer_addresses.city`. 

102 

103 Reusa a normalização de cidade da OS (`os_catalog.match_city`): caixa, 

104 acento, UF colada e os typos aprovados (`mariporã`). Cidade canônica fora 

105 das 6 (Bragança Paulista, Jarinu...) e grafia desconhecida viram 

106 "Outras cidades"; vazio vira "Cidade não informada". 

107 """ 

108 if raw is None or not raw.strip(): 

109 return CityMatch(CITY_UNKNOWN_ID, CITY_UNKNOWN_LABEL, True) 

110 canonical = os_catalog.match_city(raw) 

111 if canonical is None: 

112 return CityMatch(CITY_OTHER_ID, CITY_OTHER_LABEL, False)

114 if city_id is None: 

115 return CityMatch(CITY_OTHER_ID, CITY_OTHER_LABEL, True) 

116 return CityMatch(city_id, canonical, True)

405def monthly_goal_basis( 

406 city_id: str | None, team: str = "todos", seller_name: str | None = None 

407) -> tuple[float | None, float | None]: 

408 """ 

409 Meta e supermeta MENSAIS do recorte (`resolveGoalBasis` do portal). 

410 

411 - Corporativo -> sem meta ("Não se aplica"). 

412 - Com vendedor selecionado -> a meta individual dele (50 / 55) se for 

413 interno ativo do roster e, havendo cidade, alocado nela; senão sem meta. 

414 Comparar o realizado de um vendedor com a meta do time inteiro mostraria 

415 um dígito de atingimento para quem bateu a própria meta. 

416 - Sem cidade -> consolidado fixo 800 / 880. 

417 - Com cidade -> 50 x internos ativos alocados à cidade; sem interno 

418 (Terra Preta) -> sem meta ("Sem meta"). 

419 """ 

420 if team == "corporativo": 

421 return (None, None) 

422 if seller_name is not None: 

423 vendedor = roster_seller_by_name(seller_name) 

424 if vendedor is None or not vendedor.active or vendedor.segment != "interno": 

425 return (None, None) 

426 if city_id is not None and vendedor.city_id not in (None, city_id): 

427 return (None, None) 

428 return (INTERNAL_SELLER_MONTHLY_GOAL, super_goal_for(INTERNAL_SELLER_MONTHLY_GOAL)) 

429 if city_id is None: 

430 return (CONSOLIDATED_MONTHLY_GOAL, CONSOLIDATED_MONTHLY_SUPER_GOAL) 

431 internos = sum( 

432 1 

433 for seller in SELLER_ROSTER 

434 if seller.active and seller.segment == "interno" and seller.city_id == city_id 

435 ) 

436 if internos == 0: 

437 return (None, None) 

438 goal = INTERNAL_SELLER_MONTHLY_GOAL * internos 

439 return (goal, super_goal_for(goal))
