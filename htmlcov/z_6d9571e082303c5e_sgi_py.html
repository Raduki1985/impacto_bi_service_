1""" 

2Conexão somente leitura com o SGI, o PostgreSQL do ERP que é a origem dos dados 

3de BI. 

4 

5É um segundo engine, separado de `app.db.database` (o banco do próprio serviço: 

6usuários, sessões, notificações). As diferenças são de propósito: 

7 

8- **Somente leitura**: a sessão abre com `default_transaction_read_only=on`, então 

9 qualquer INSERT/UPDATE/DELETE falha no servidor mesmo que o usuário do banco 

10 tivesse permissão. O serviço nunca escreve no ERP. 

11- **Pool real**: as consultas de BI são frequentes e curtas; abrir uma conexão 

12 TCP por request (o `NullPool` do banco do serviço) custaria mais do que a 

13 própria query. 

14- **`statement_timeout`**: o SGI é o banco transacional da operação; uma 

15 consulta pesada do BI não pode ficar presa lá. 

16 

17Uso (SQLAlchemy Core, sem ORM: as consultas são SQL parametrizado com `text()`): 

18 

19 with sgi_connection() as conn: 

20 linhas = conn.execute(text("SELECT ... WHERE x = :x"), {"x": 1}).mappings().all() 

21 

22Sem `SGI_DATABASE_URL` configurada, `sgi_connection()` levanta 

23`ServiceUnavailableError` (503) dizendo o que falta. 

24"""

47def sgi_configured() -> bool: 

48 """`SGI_DATABASE_URL` preenchida? Sem ela a integração fica desligada.""" 

49 return bool(settings.SGI_DATABASE_URL.strip())

52def connect_args_for(url: str, query_timeout: int) -> dict[str, object]: 

53 """ 

54 Opções de sessão passadas ao driver. 

55 

56 Só valem para PostgreSQL (psycopg2 ou psycopg 3, ambos via libpq). Qualquer 

57 outra URL (um SQLite de teste, por exemplo) recebe um dicionário vazio. 

58 """ 

59 if not url.startswith("postgresql"): 

60 return {} 

61 return { 

62 "application_name": APPLICATION_NAME, 

63 "connect_timeout": 10, 

64 "options": ( 

65 "-c default_transaction_read_only=on " 

66 f"-c statement_timeout={query_timeout * 1000} " 

67 "-c TimeZone=America/Sao_Paulo" 

68 ), 

69 }

72def build_engine(url: str, query_timeout: int) -> Engine: 

73 """Engine com pool e opções de sessão; não abre conexão nenhuma ainda.""" 

74 kwargs: dict[str, object] = {"pool_pre_ping": True} 

75 if url.startswith("postgresql"): 

76 kwargs.update( 

77 pool_size=5, 

78 max_overflow=5, 

79 pool_recycle=1800, 

80 connect_args=connect_args_for(url, query_timeout), 

81 ) 

82 return create_engine(url, **kwargs)

85def get_sgi_engine() -> Engine: 

86 global _engine 

87 if not sgi_configured(): 

88 raise ServiceUnavailableError( 

89 "Conexão com o SGI não configurada: defina SGI_DATABASE_URL no .env." 

90 ) 

91 if _engine is None: 

92 _engine = build_engine(settings.SGI_DATABASE_URL, settings.SGI_QUERY_TIMEOUT) 

93 return _engine

96def reset_sgi_engine() -> None: 

97 """Descarta o engine memoizado (testes e troca de configuração)."""

104def _causa(e: DBAPIError) -> str: 

105 origem = e.orig if e.orig is not None else e 

106 linhas = str(origem).strip().splitlines() 

107 return linhas[0] if linhas else str(origem)

110@contextmanager 

111def sgi_connection() -> Iterator[Connection]: 

112 """ 

113 Conexão do pool, já em transação somente leitura. 

114 

115 Falhas de rede, credencial ou da própria consulta viram 

116 `ServiceUnavailableError` (503) com a primeira linha da causa na mensagem: 

117 para quem está configurando a integração, o motivo importa. 

118 """ 

119 engine = get_sgi_engine() 

120 try: 

121 with engine.connect() as conn: 

122 yield conn 

123 except DBAPIError as e: 

124 logger.warning("SGI: falha na conexão ou na consulta: %s", _causa(e)) 

125 raise ServiceUnavailableError(f"Falha ao consultar o SGI: {_causa(e)}") from e 

126 except SQLAlchemyError as e: 

127 logger.exception("SGI: erro inesperado") 

128 raise ServiceUnavailableError(f"Falha ao consultar o SGI: {e}") from e

131@dataclass(frozen=True) 

132class SgiConnectionInfo: 

133 """Resultado do teste de conexão.""" 

134 

135 database: str 

136 user: str 

137 server_version: str 

138 server_time: datetime 

139 read_only: bool

141 

142_PING = text( 

143 "SELECT current_database() AS banco, " 

144 "current_user AS usuario, " 

145 "current_setting('server_version') AS versao, " 

146 "now() AS agora, " 

147 "current_setting('default_transaction_read_only') AS somente_leitura" 

148) 

149 

150 

151def check_sgi_connection() -> SgiConnectionInfo: 

152 """ 

153 Confirma que dá para conectar e consultar, e que a sessão é somente leitura. 

154 

155 Levanta `ServiceUnavailableError` quando falha, com a causa na mensagem. 

156 """ 

157 with sgi_connection() as conn: 

158 linha = conn.execute(_PING).mappings().one() 

159 return SgiConnectionInfo( 

160 database=str(linha["banco"]), 

161 user=str(linha["usuario"]), 

162 server_version=str(linha["versao"]), 

163 server_time=linha["agora"], 

164 read_only=str(linha["somente_leitura"]).lower() == "on", 

165 )
